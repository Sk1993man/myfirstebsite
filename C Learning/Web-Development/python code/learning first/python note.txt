# इस टूल ब्लॉक का उपयोग करके हम यूजर के लिए 'Developer_Learning_Planner.pdf' नाम की एक फाइल जेनेरेट कर रहे हैं।
import os
from reportlab.lib.pagesizes import letter
from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Table, TableStyle
from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
from reportlab.lib import colors

def create_planner():
    pdf_filename = "Developer_Learning_Planner.pdf"
    doc = SimpleDocTemplate(pdf_filename, pagesize=letter, rightMargin=40, leftMargin=40, topMargin=40, bottomMargin=40)
    story = []
    styles = getSampleStyleSheet()
    
    # Custom Styles
    title_style = ParagraphStyle('Title', parent=styles['Heading1'], fontSize=22, textColor=colors.HexColor('#dc3545'), spaceAfter=15, alignment=1)
    heading_style = ParagraphStyle('Heading', parent=styles['Heading2'], fontSize=14, textColor=colors.HexColor('#222222'), spaceBefore=12, spaceAfter=8)
    body_style = ParagraphStyle('Body', parent=styles['Normal'], fontSize=10, leading=14, textColor=colors.HexColor('#444444'))
    
    # Content
    story.append(Paragraph("<b>SAURABH'S WEB DEVELOPER LEARNING PLANNER</b>", title_style))
    story.append(Paragraph("A step-by-step roadmap to go from a beginner to a professional full-stack developer.", body_style))
    story.append(Spacer(1, 15))
    
    # Table Data
    data = [
        [Paragraph("<b>Timeline</b>", body_style), Paragraph("<b>Topics to Learn</b>", body_style), Paragraph("<b>Daily Practical Task</b>", body_style)],
        [Paragraph("Month 1", body_style), Paragraph("HTML5 & CSS3 Basics (Tags, Layouts, Forms)", body_style), Paragraph("Build 3 simple landing pages from scratch.", body_style)],
        [Paragraph("Month 2", body_style), Paragraph("Advanced CSS, Flexbox, Mobile Responsiveness", body_style), Paragraph("Make your previous pages perfectly fit on mobile screens.", body_style)],
        [Paragraph("Month 3", body_style), Paragraph("JavaScript Basics & Logical Thinking (Variables, Functions)", body_style), Paragraph("Create interactive elements like calculators or pop-ups.", body_style)],
        [Paragraph("Month 4", body_style), Paragraph("Advanced JS (DOM Manipulation) + Continue C Language", body_style), Paragraph("Connect logic with your HTML pages. Practice loops in C.", body_style)],
        [Paragraph("Month 5", body_style), Paragraph("Git, GitHub, Frontend Frameworks (React)", body_style), Paragraph("Upload every single project to GitHub Desktop.", body_style)],
        [Paragraph("Month 6", body_style), Paragraph("Backend (Node.js) & Database (MongoDB)", body_style), Paragraph("Build a complete full-stack website with login system.", body_style)]
    ]
    
    t = Table(data, colWidths=[60, 220, 240])
    t.setStyle(TableStyle([
        ('BACKGROUND', (0,0), (-1,0), colors.HexColor('#f8f9fa')),
        ('TEXTCOLOR', (0,0), (-1,0), colors.HexColor('#212529')),
        ('ALIGN', (0,0), (-1,-1), 'LEFT'),
        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
        ('BOTTOMPADDING', (0,0), (-1,0), 8),
        ('BACKGROUND', (0,1), (-1,-1), colors.white),
        ('GRID', (0,0), (-1,-1), 0.5, colors.HexColor('#dee2e6')),
        ('TOPPADDING', (0,0), (-1,-1), 8),
        ('BOTTOMPADDING', (0,0), (-1,-1), 8),
    ]))
    
    story.append(t)
    story.append(Spacer(1, 15))
    story.append(Paragraph("<b>Essential Tool Checklist:</b>", heading_style))
    story.append(Paragraph("1. <b>VS Code</b> - For writing all code.<br/>2. <b>GitHub Desktop</b> - For uploading and tracking progress.<br/>3. <b>Google Chrome DevTools</b> - For testing website designs directly.", body_style))
    
    doc.build(story)

create_planner()
